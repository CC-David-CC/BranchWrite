```python
from reportlab.lib import colors
from reportlab.lib.pagesizes import A4
from reportlab.lib.styles import getSampleStyleSheet
from reportlab.platypus import SimpleDocTemplate, Paragraph, Spacer, Table, TableStyle, PageBreak

styles = getSampleStyleSheet()
doc = SimpleDocTemplate('document.pdf', pagesize=A4,
    leftMargin=48, rightMargin=48, topMargin=54, bottomMargin=48)

def page_number(canvas, document):
    canvas.saveState()
    canvas.setFont('Helvetica', 10)
    canvas.drawRightString(A4[0] - 48, 28, 'Page ' + str(document.page))
    canvas.restoreState()

story = [Paragraph('Before the visit', styles['Heading1']),
    Paragraph('Prepare for a visit to your local library.', styles['BodyText']),
    Spacer(1, 12)]
rows = [[Paragraph('Item', styles['BodyText']), Paragraph('Action', styles['BodyText'])],
    [Paragraph('Library card', styles['BodyText']),
     Paragraph('Bring your library card. Check the library opening hours before leaving home.', styles['BodyText'])],
    [Paragraph('Book list', styles['BodyText']),
     Paragraph('Write a short list of books to find, and ask a librarian for help if needed.', styles['BodyText'])]]
table = Table(rows, colWidths=[110, doc.width - 110])
table.setStyle(TableStyle([
    ('BACKGROUND', (0, 0), (-1, 0), colors.lightgrey),
    ('GRID', (0, 0), (-1, -1), 0.5, colors.grey),
    ('VALIGN', (0, 0), (-1, -1), 'TOP'),
    ('LEFTPADDING', (0, 0), (-1, -1), 8),
    ('RIGHTPADDING', (0, 0), (-1, -1), 8),
    ('TOPPADDING', (0, 0), (-1, -1), 8),
    ('BOTTOMPADDING', (0, 0), (-1, -1), 8)]))
story.append(table)
story.append(PageBreak())
story.append(Paragraph('After the visit', styles['Heading1']))
story.append(Paragraph('Keep borrowed books in a safe place and record their return dates.', styles['BodyText']))
doc.build(story, onFirstPage=page_number, onLaterPages=page_number)
```
